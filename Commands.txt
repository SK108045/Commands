FIREWALL RULES "
$InstallationDir = 'C:\eTIMS-Paypoint-Win'

$Rules = @(
    @{
        Name = 'Codex-eTIMS-Research-Block-Client'
        Program = Join-Path $InstallationDir 'bin\EBM2x.WPF.exe'
    },
    @{
        Name = 'Codex-eTIMS-Research-Block-Launcher'
        Program = Join-Path $InstallationDir 'EBM2xLauncher.exe'
    }
)

foreach ($Rule in $Rules) {
    if (-not (Test-Path -LiteralPath $Rule.Program -PathType Leaf)) {
        throw "eTIMS executable not found: $($Rule.Program)"
    }

    $Existing = Get-NetFirewallRule -Name $Rule.Name `
        -ErrorAction SilentlyContinue

    if ($Existing) {
        Write-Host "Rule already exists: $($Rule.Name)" `
            -ForegroundColor Yellow
        continue
    }

    New-NetFirewallRule `
        -Name $Rule.Name `
        -DisplayName $Rule.Name `
        -Enabled True `
        -Direction Outbound `
        -Action Block `
        -Profile Any `
        -Program $Rule.Program `
        -Protocol Any `
        -PolicyStore PersistentStore

    Write-Host "Created: $($Rule.Name)" -ForegroundColor Green
}

Write-Host "`n=== VERIFY RULES ===" -ForegroundColor Cyan

Get-NetFirewallRule `
    -Name 'Codex-eTIMS-Research-Block-Client',
          'Codex-eTIMS-Research-Block-Launcher' |
    Format-Table Name, Enabled, Direction, Action, Profile
